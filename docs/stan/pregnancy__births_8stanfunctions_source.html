48vector compute_baseline_birth_rate(
49 real birth_rate_baseline_min,
50 real birth_rate_baseline_max,
51 real herring_birth_rate_midpoint,
52 real herring_slope,
53 real herring_weight,
54 data vector herring_index_baltic_proper_gulf_finland,
55 data vector herring_index_gulf_bothnia
56) {
57 int n_years = num_elements(herring_index_baltic_proper_gulf_finland);
58 vector[n_years] birth_rate_baseline;
59
60 for (t in 1:n_years) {
61 real weighted_h =
62 herring_weight * herring_index_baltic_proper_gulf_finland[t]
63 + (1.0 - herring_weight) * herring_index_gulf_bothnia[t];
64
65 birth_rate_baseline[t] =
66 birth_rate_baseline_min
67 + (birth_rate_baseline_max - birth_rate_baseline_min)
68 * inv_logit(
69 herring_slope
70 * (weighted_h - herring_birth_rate_midpoint)
71 );
72 }
73
74 return birth_rate_baseline;
75}

110real compute_density_dependence_intercept(
111 real birth_rate_baseline_max,
112 real density_dependence_scaled
113) {
114 real complement =
115 (1.0 - density_dependence_scaled)
116 * (1.0 - birth_rate_baseline_max);
117
118 // Use log1p near an argument of one, preserving the small intercept.
119 if (complement < 0.5) {
120 return -log1p(-complement);
121 }
122
123 // Near an argument of zero, avoid subtracting the complement from one.
124 return -log(
125 birth_rate_baseline_max
126 + density_dependence_scaled * (1.0 - birth_rate_baseline_max)
127 );
128}

199real update_birth_rate(
200 real birth_rate_baseline,
201 real density_dependence_intercept,
202 real density_dependence_slope,
203 real population_total_previous
204) {
205 // Density-dependent exponent and birth penalty
206 real density_exponent = population_total_previous * density_dependence_slope;
207 real density_penalty = 0.0;
208
209 // Calculate the penalty without cancellation or intermediate overflow
210 if (density_dependence_intercept != 0.0 && density_exponent != 0.0) {
211 real log_density_increase = density_exponent <= log(2.0)
212 ? log(expm1(density_exponent))
213 : density_exponent + log1m_exp(-density_exponent);
214 density_penalty = exp(log(density_dependence_intercept) + log_density_increase);
215 }
216
217 // Apply density suppression to the baseline birth rate
218 return exp(log(birth_rate_baseline) - density_penalty);
219}
