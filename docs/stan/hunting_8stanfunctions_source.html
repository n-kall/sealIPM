55vector compute_hunting_pressure(
56 data int hunting_quota,
57 vector hunting_selectivity,
58 vector population_comp,
59 real hunting_effort_sd,
60 real hunting_effort_noise,
61 data real hunting_duration
62) {
63 int n_demo_groups = num_elements(population_comp);
64 vector[n_demo_groups] pressure;
65
66 if (hunting_quota == 0) {
67 return rep_vector(0.0, n_demo_groups);
68 }
69
70 if (!(hunting_duration > 0.0)) {
71 reject(
72 "compute_hunting_pressure requires ",
73 "hunting_duration > 0; received ", hunting_duration
74 );
75 }
76
77 if (!(min(population_comp) >= 0.0)) {
78 reject(
79 "compute_hunting_pressure requires non-negative counts; ",
80 "population_comp=", population_comp
81 );
82 }
83
84 if (!(sum(population_comp) > 0.0)) {
85 reject(
86 "compute_hunting_pressure requires a positive total ",
87 "population when the quota is positive."
88 );
89 }
90
91 {
92 vector[n_demo_groups] log_population = log(population_comp);
93 real log_denominator =
94 log_sum_exp(hunting_selectivity + log_population);
95
96 pressure = exp(
97 hunting_selectivity
98 + log(hunting_quota)
99 + log(2.0)
100 - 2.0 * log(hunting_duration)
101 - hunting_effort_noise * hunting_effort_sd
102 - log_denominator
103 );
104 }
105
106 for (demo in 1:n_demo_groups) {
107 if (population_comp[demo] == 0.0) {
108 pressure[demo] = 0.0;
109 }
110 }
111
112 return pressure;
113}

162vector hunting_count_derivative(
163 real time_since_hunting_start,
164 vector cumulative_hunting_count,
165 real population_at_hunting_start,
166 real hunting_duration,
167 real hunting_pressure_focal,
168 real hunting_pressure_competing,
169 real non_hunting_mortality_rate
170) {
171 // Expected population remaining during the hunting interval
172 real expected_alive_population =
173 population_at_hunting_start
174 * exp(
175 -(hunting_pressure_focal + hunting_pressure_competing)
176 * (hunting_duration * time_since_hunting_start - square(time_since_hunting_start) / 2.0)
177 - non_hunting_mortality_rate * time_since_hunting_start
178 );
179
180 // Instantaneous hunting count for the focal country
181 real hunting_count_rate =
182 expected_alive_population
183 * hunting_pressure_focal
184 * (hunting_duration - time_since_hunting_start);
185
186 return rep_vector(hunting_count_rate, 1);
187}

298vector compute_fate_probabilities(
299 data real t_birth_to_start_hunt,
300 data real hunting_duration,
301 real hunting_pressure_sweden,
302 real hunting_pressure_finland,
303 real non_hunting_mortality_rate,
304 data real rel_tol,
305 data real abs_tol,
306 data int max_num_steps,
307 data real probability_tolerance
308) {
309 vector[4] probability;
310 real probability_sum;
311 real k = hunting_duration;
312 real survival_to_hunting_start =
313 exp(-non_hunting_mortality_rate * t_birth_to_start_hunt);
314 vector[1] initial_hunting = rep_vector(0.0, 1);
315 array[1] real times = { k };
316
317 if (!(k >= 0.0 && k <= 1.0)) {
318 reject(
319 "compute_fate_probabilities requires hunting_duration in [0, 1]; ",
320 "received ", k
321 );
322 }
323 if (!(t_birth_to_start_hunt >= 0.0 && t_birth_to_start_hunt <= 1.0 - k)) {
324 reject(
325 "compute_fate_probabilities requires birth-to-hunting time in [0, 1 - k]; ",
326 "received ", t_birth_to_start_hunt, ", hunting duration=", k
327 );
328 }
329 if (
330 !(hunting_pressure_sweden >= 0.0)
331 || !(hunting_pressure_finland >= 0.0)
332 || !(non_hunting_mortality_rate >= 0.0)
333 ) {
334 reject(
335 "compute_fate_probabilities requires non-negative hazards; ",
336 "hunting_pressure_sweden=", hunting_pressure_sweden,
337 ", hunting_pressure_finland=", hunting_pressure_finland,
338 ", non_hunting_mortality_rate=", non_hunting_mortality_rate
339 );
340 }
341
342 // Integrate Swedish hunting and account for survival to hunting onset
343 probability[2] = 0.0;
344 if (k > 0.0 && hunting_pressure_sweden > 0.0) {
345 array[1] vector[1] hunted_sweden = ode_rk45_tol(
346 hunting_count_derivative, initial_hunting, 0.0, times,
347 rel_tol, abs_tol, max_num_steps,
348 1.0, k, hunting_pressure_sweden, hunting_pressure_finland, non_hunting_mortality_rate
349 );
350 probability[2] = survival_to_hunting_start * hunted_sweden[1][1];
351 }
352
353 // Integrate Finnish hunting and account for survival to hunting onset
354 probability[3] = 0.0;
355 if (k > 0.0 && hunting_pressure_finland > 0.0) {
356 array[1] vector[1] hunted_finland = ode_rk45_tol(
357 hunting_count_derivative, initial_hunting, 0.0, times,
358 rel_tol, abs_tol, max_num_steps,
359 1.0, k, hunting_pressure_finland, hunting_pressure_sweden, non_hunting_mortality_rate
360 );
361 probability[3] = survival_to_hunting_start * hunted_finland[1][1];
362 }
363
364 // Calculate survival and remaining non-hunting death probability
365 probability[1] =
366 exp(-(hunting_pressure_sweden + hunting_pressure_finland) * square(k) / 2.0)
367 * exp(-non_hunting_mortality_rate);
368 probability[4] =
369 1.0 - probability[1] - probability[2] - probability[3];
370
371 // Validate the probability simplex before correcting small numerical errors
372 probability_sum =
373 sum(probability);
374
375 if (!(probability_tolerance > 0.0)) {
376 reject(
377 "compute_fate_probabilities requires a positive tolerance; ",
378 "probability_tolerance=", probability_tolerance
379 );
380 }
381
382 /*
383 * The !(x >= limit) form also rejects NaN because comparisons involving
384 * NaN evaluate to false
385 */
386 if (!(min(probability) >= -probability_tolerance)) {
387 reject(
388 "Joint fate calculation returned a materially negative probability; ",
389 "probability=", probability,
390 ", tolerance=", probability_tolerance
391 );
392 }
393
394 if (
395 !(probability_sum >= 1.0 - probability_tolerance)
396 || !(probability_sum <= 1.0 + probability_tolerance)
397 ) {
398 reject(
399 "Joint fate calculation failed probability conservation; probability=",
400 probability,
401 ", sum=", probability_sum,
402 ", tolerance=", probability_tolerance
403 );
404 }
405
406 /*
407 * Clamp only small boundary violations that passed the tolerance check
408 */
409 for (fate_id in 1:num_elements(probability)) {
410 probability[fate_id] =
411 fmax(0.0, probability[fate_id]);
412 }
413
414 probability_sum =
415 sum(probability);
416
417 if (!(probability_sum > 0.0)) {
418 reject(
419 "Joint fate probabilities sum to zero or are non-finite; probability=",
420 probability
421 );
422 }
423
424 /*
425 * Restore an exact simplex after any tolerated boundary correction
426 */
427 return probability / probability_sum;
428}
