19 data vector population_init,
20 real population_init_size,
21 data int population_burn_in,
22 real birth_rate_year,
23 data matrix aging_matrix,
24 vector survival_probability,
25 data int n_age_classes
26) {
27 // Demographic dimensions
28 int n_demo_groups = 2 * n_age_classes;
29
30 // Population state during burn-in
31 vector[n_demo_groups] population_comp_year = population_init;
32 real population_total_year;
33
34 // Run deterministic survival, aging and births to obtain the initial composition
35 for (burn_in_step in 1:population_burn_in) {
36 // Apply survival and advance seals through the age classes
37 population_comp_year =
38 aging_matrix
39 * diag_matrix(survival_probability)
40 * population_comp_year;
41
42 // Add female pups using the adult female population
43 population_comp_year[1] =
44 birth_rate_year / 2.0
45 * population_comp_year[n_age_classes];
46
47 // Add the same expected number of male pups
48 population_comp_year[n_age_classes + 1] =
49 birth_rate_year / 2.0
50 * population_comp_year[n_age_classes];
51 }
52
53 // Rescale the demographic composition to the target initial population size
54 population_comp_year =
55 population_comp_year
56 * population_init_size
57 / sum(population_comp_year);
58
59 // Calculate the total population after rescaling
60 population_total_year = sum(population_comp_year);
61
62 // Return the initial composition, birth rate and population total
63 return (
64 population_comp_year,
65 birth_rate_year,
66 population_total_year
67 );
68}

340 * placental_scar_detection_probability[year]

342

344 birth_rate[year]
345 * (1.0 - placental_scar_detection_probability[year])
346 * ca_detection_probability[year]

349 * ca_detection_probability[year];
350

353 * placental_scar_detection_probability[year]

355
356 reproductive_probs[1, year] =

377}
