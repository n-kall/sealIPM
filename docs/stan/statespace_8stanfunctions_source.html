38vector compute_baseline_birth_rate(
39 real birth_rate_baseline_min,
40 real birth_rate_baseline_max,
41 real herring_intercept_scaled,
42 real herring_slope,
43 real herring_weight,
44 data vector herring_index_baltic_proper_gulf_finland,
45 data vector herring_index_gulf_bothnia
46) {
47 int n_years = num_elements(herring_index_baltic_proper_gulf_finland);
48 vector[n_years] birth_rate_baseline;
49
50 for (t in 1:n_years) {
51 real weighted_h =
52 herring_weight * herring_index_baltic_proper_gulf_finland[t]
53 + (1.0 - herring_weight) * herring_index_gulf_bothnia[t];
54
55 birth_rate_baseline[t] =
56 birth_rate_baseline_min
57 + (birth_rate_baseline_max - birth_rate_baseline_min)
58 * inv_logit(
59 herring_slope
60 * (herring_intercept_scaled + weighted_h)
61 );
62 }
63
64 return birth_rate_baseline;
65}

99real compute_density_dependence_intercept(
100 real birth_rate_baseline_max,
101 real density_dependence_scaled
102) {
103 real complement =
104 (1.0 - density_dependence_scaled)
105 * (1.0 - birth_rate_baseline_max);
106
107 // Use log1p near an argument of one, preserving the small intercept.
108 if (complement < 0.5) {
109 return -log1p(-complement);
110 }
111
112 // Near an argument of zero, avoid subtracting the complement from one.
113 return -log(
114 birth_rate_baseline_max
115 + density_dependence_scaled * (1.0 - birth_rate_baseline_max)
116 );
117}

186real update_birth_rate(
187 real birth_rate_baseline,
188 real density_dependence_intercept,
189 real density_dependence_slope,
190 real population_total_previous
191) {
192 // Density-dependent exponent and birth penalty
193 real density_exponent = population_total_previous * density_dependence_slope;
194 real density_penalty = 0.0;
195
196 // Calculate the penalty without cancellation or intermediate overflow
197 if (density_dependence_intercept != 0.0 && density_exponent != 0.0) {
198 real log_density_increase = density_exponent <= log(2.0)
199 ? log(expm1(density_exponent))
200 : density_exponent + log1m_exp(-density_exponent);
201 density_penalty = exp(log(density_dependence_intercept) + log_density_increase);
202 }
203
204 // Apply density suppression to the baseline birth rate
205 return exp(log(birth_rate_baseline) - density_penalty);
206}

352vector mortality_rates(
353 real female_pup_survival_probability,
354 real female_adult_survival_probability,
355 real mortality_age_shape,
356 data int n_age_classes,
357 real male_pup_offset,
358 real male_adult_offset
359) {
360 int n_demo = 2 * n_age_classes;
361 vector[n_demo] mu_m;
362 real mu_pup_f = -log(female_pup_survival_probability);
363 real mu_ad_f = -log(female_adult_survival_probability);
364 real mu_pup_m = exp(log(mu_pup_f) + male_pup_offset);
365 real mu_ad_m = exp(log(mu_ad_f) + male_adult_offset);
366
367 mu_m[1] = mu_pup_f;
368 mu_m[n_age_classes] = mu_ad_f;
369
370 for (j in 2:(n_age_classes - 1)) {
371 real w = ((j - 1.0) / (n_age_classes - 1.0))^mortality_age_shape;
372
373 mu_m[j] = exp(
374 log(mu_pup_f)
375 + w * (log(mu_ad_f) - log(mu_pup_f))
376 );
377 }
378
379 mu_m[n_age_classes + 1] = mu_pup_m;
380 mu_m[n_demo] = mu_ad_m;
381
382 for (j in 2:(n_age_classes - 1)) {
383 real w = ((j - 1.0) / (n_age_classes - 1.0))^mortality_age_shape;
384
385 mu_m[n_age_classes + j] = exp(
386 log(mu_pup_m)
387 + w * (log(mu_ad_m) - log(mu_pup_m))
388 );
389 }
390
391 return mu_m;
392}

435vector update_population_from_survivors(
436 vector previous_survivors,
437 data matrix aging_mat,
438 real birth_rate,
439 real birth_count_noise,
440 real pup_sex_allocation_noise,
441 data int n_age_classes
442) {
443 int n_demo = rows(previous_survivors);
444 vector[n_demo] N = aging_mat * previous_survivors;
445 real N_temp;
446
447 N_temp =
448 N[n_age_classes] * birth_rate
449 + sqrt(
450 N[n_age_classes]
451 * birth_rate
452 * (1.0 - birth_rate)
453 ) * birth_count_noise;
454
455 N[1] =
456 N_temp / 2.0
457 + sqrt(N_temp / 4.0) * pup_sex_allocation_noise;
458
459 N[n_age_classes + 1] =
460 N_temp - N[1];
461
462 return N;
463}

477tuple(vector, real, real) initialize_population_with_burnin(
478 data vector population_init,
479 real population_init_size,
480 data int population_burn_in,
481 real birth_rate_year,
482 data matrix aging_matrix,
483 vector survival_probability,
484 data int n_age_classes
485) {
486 // Demographic dimensions
487 int n_demo_groups = 2 * n_age_classes;
488
489 // Population state during burn-in
490 vector[n_demo_groups] population_comp_year = population_init;
491 real population_total_year;
492
493 // Run deterministic survival, aging and births to obtain the initial composition
494 for (burn_in_step in 1:population_burn_in) {
495 // Apply survival and advance seals through the age classes
496 population_comp_year =
497 aging_matrix
498 * diag_matrix(survival_probability)
499 * population_comp_year;
500
501 // Add female pups using the adult female population
502 population_comp_year[1] =
503 birth_rate_year / 2.0
504 * population_comp_year[n_age_classes];
505
506 // Add the same expected number of male pups
507 population_comp_year[n_age_classes + 1] =
508 birth_rate_year / 2.0
509 * population_comp_year[n_age_classes];
510 }
511
512 // Rescale the demographic composition to the target initial population size
513 population_comp_year =
514 population_comp_year
515 * population_init_size
516 / sum(population_comp_year);
517
518 // Calculate the total population after rescaling
519 population_total_year = sum(population_comp_year);
520
521 // Return the initial composition, birth rate and population total
522 return (
523 population_comp_year,
524 birth_rate_year,
525 population_total_year
526 );
527}

569vector compute_hunting_pressure(
570 data int hunting_quota,
571 vector hunting_selectivity,
572 vector population_comp,
573 real hunting_effort_sd,
574 real hunting_effort_noise,
575 data real t_birth_to_end_hunt
576) {
577 int n_demo_groups = num_elements(population_comp);
578
579 if (hunting_quota == 0) {
580 return rep_vector(0.0, n_demo_groups);
581 }
582
583 if (!(t_birth_to_end_hunt > 0.0)) {
584 reject(
585 "compute_hunting_pressure requires t_birth_to_end_hunt > 0; ",
586 "received ", t_birth_to_end_hunt
587 );
588 }
589
590 if (!(min(population_comp) > 0.0)) {
591 reject(
592 "compute_hunting_pressure requires strictly positive population ",
593 "counts; population_comp=", population_comp
594 );
595 }
596
597 {
598 vector[n_demo_groups] log_population = log(population_comp);
599 real log_denominator =
600 log_sum_exp(hunting_selectivity + log_population);
601
602 return exp(
603 hunting_selectivity
604 + log(hunting_quota)
605 + log(2.0)
606 - 2.0 * log(t_birth_to_end_hunt)
607 - hunting_effort_noise * hunting_effort_sd
608 - log_denominator
609 );
610 }

790row_vector allocate_fates(
791 row_vector expected_fate_counts,
792 row_vector fate_allocation_noise,
793 real demographic_population_count,
794 int hunting_active_sweden,
795 int hunting_active_finland
796) {
797 // Active category dimensions and positions in paper order
798 int n_active_fates = 2 + hunting_active_sweden + hunting_active_finland;
799 array[n_active_fates] int active_fate_indices;
800 int next_active_fate = 2;
801
802 // Expected counts and adjusted concentrations
803 vector[n_active_fates] expected_active_fate_counts;
804 vector[n_active_fates] concentration_adjusted;
805 real min_expected_count;
806
807 // Log-ratio means and structured covariance factor
808 vector[n_active_fates - 1] mean_logratio;
809 vector[n_active_fates - 1] diagonal_variance;
810 real shared_variance;
811 matrix[n_active_fates - 1, n_active_fates - 1] fate_allocation_cholesky =
812 rep_matrix(0.0, n_active_fates - 1, n_active_fates - 1);
813
814 // Active allocations and full four-category output
815 vector[n_active_fates] logits;
816 vector[n_active_fates] active_fate_counts;
817 row_vector[4] realized_fate_counts = rep_row_vector(0.0, 4);
818
819 // Check dimensions and the population count before allocating
820 if (num_elements(expected_fate_counts) != 4) {
821 reject("allocate_fates requires four expected fate counts");
822 }
823 if (num_elements(fate_allocation_noise) < n_active_fates - 1) {
824 reject("allocate_fates requires at least ", n_active_fates - 1,
825 " innovations, received ", num_elements(fate_allocation_noise));
826 }
827 if (!(demographic_population_count > 0.0)) {
828 reject("allocate_fates requires a positive population count; received ",
829 demographic_population_count);
830 }
831
832 // Select survival, active hunting categories, and other death or bycatch
833 active_fate_indices[1] = 1;
834 if (hunting_active_sweden) {
835 active_fate_indices[next_active_fate] = 2;
836 next_active_fate += 1;
837 }
838 if (hunting_active_finland) {
839 active_fate_indices[next_active_fate] = 3;
840 next_active_fate += 1;
841 }
842 active_fate_indices[n_active_fates] = 4;
843 expected_active_fate_counts = expected_fate_counts[active_fate_indices]';
844 min_expected_count = min(expected_active_fate_counts);
845 if (!(min_expected_count > 0.0)) {
846 reject("allocate_fates requires strictly positive active expected counts; received ",
847 expected_active_fate_counts);
848 }
849
850 // Adjust concentrations without forming a potentially overflowing reciprocal
851 concentration_adjusted = expected_active_fate_counts
852 + expected_active_fate_counts / min_expected_count;
853
854 // Calculate log-ratio means and variances relative to survival
855 mean_logratio = digamma(concentration_adjusted[2:n_active_fates])
856 - rep_vector(digamma(concentration_adjusted[1]), n_active_fates - 1);
857 diagonal_variance = trigamma(concentration_adjusted[2:n_active_fates]);
858 shared_variance = trigamma(concentration_adjusted[1]);
859
860 // Factor diag(diagonal_variance) + shared_variance * 1 * 1' directly
861 // This avoids losing small diagonal terms when the shared variance is large
862 for (i in 1:(n_active_fates - 1)) {
863 real pivot_variance = diagonal_variance[i] + shared_variance;
864 fate_allocation_cholesky[i, i] = sqrt(pivot_variance);
865 if (i < n_active_fates - 1) {
866 for (j in (i + 1):(n_active_fates - 1)) {
867 fate_allocation_cholesky[j, i] = shared_variance / fate_allocation_cholesky[i, i];
868 }
869 // A bounded ratio avoids cancellation and overflow in the variance update
870 shared_variance *= diagonal_variance[i] / pivot_variance;
871 }
872 }
873
874 // Apply the same innovations and convert log-ratios into population counts
875 logits[1] = 0.0;
876 logits[2:n_active_fates] = mean_logratio
877 + fate_allocation_cholesky * fate_allocation_noise[1:(n_active_fates - 1)]';
878 active_fate_counts = demographic_population_count * softmax(logits);
879
880 // Return counts in paper order with inactive hunting categories left at zero
881 for (i in 1:n_active_fates) {
882 realized_fate_counts[active_fate_indices[i]] = active_fate_counts[i];
883 }
884
885 return realized_fate_counts;
886}

1094 transition_noise_year =
1095 to_matrix(
1096 transition_noise_raw[, year],
1097 n_demo_groups,

1099 );

1103 int hunting_active_sweden = hunting_quota_sweden[year] > 0;
1104 int hunting_active_finland = hunting_quota_finland[year] > 0;
1105

1110 expected_fate_counts[demo, ],

1114 hunting_active_finland
1115 );
1116 }
