159vector mortality_rates(
160 real phi_pup,
161 real phi_adult,
162 real c,
163 data int n_age_classes,
164 real male_pup_offset,
165 real male_adult_offset
166) {
167 int n_demo = 2 * n_age_classes;
168 vector[n_demo] mu_m;
169 real mu_pup_f = -log(phi_pup);
170 real mu_ad_f = -log(phi_adult);
171 real mu_pup_m = exp(log(mu_pup_f) + male_pup_offset);
172 real mu_ad_m = exp(log(mu_ad_f) + male_adult_offset);
173
174 mu_m[1] = mu_pup_f;
175 mu_m[n_age_classes] = mu_ad_f;
176
177 for (j in 2:(n_age_classes - 1)) {
178 real w = ((j - 1.0) / (n_age_classes - 1.0))^c;
179
180 mu_m[j] = exp(
181 log(mu_pup_f)
182 + w * (log(mu_ad_f) - log(mu_pup_f))
183 );
184 }
185
186 mu_m[n_age_classes + 1] = mu_pup_m;
187 mu_m[n_demo] = mu_ad_m;
188
189 for (j in 2:(n_age_classes - 1)) {
190 real w = ((j - 1.0) / (n_age_classes - 1.0))^c;
191
192 mu_m[n_age_classes + j] = exp(
193 log(mu_pup_m)
194 + w * (log(mu_ad_m) - log(mu_pup_m))
195 );
196 }
197
198 return mu_m;
199}

340 data int hunting_quota,

342 vector population_comp,
343 real hunting_effort_sd,
344 real epsilon_h,
345 data real t_birth_to_end_hunt
346) {
347 int n_demo_groups = num_elements(population_comp);
348
349 if (hunting_quota == 0) {
350 return rep_vector(0.0, n_demo_groups);
351 }
352
353 if (!(t_birth_to_end_hunt > 0.0)) {
354 reject(
355 "compute_hunting_pressure requires t_birth_to_end_hunt > 0; ",
356 "received ", t_birth_to_end_hunt
357 );
358 }
359

361 reject(
362 "compute_hunting_pressure requires strictly positive population ",

364 );

366

368 vector[n_demo_groups] log_population = log(population_comp);
369 real log_denominator =
370 log_sum_exp(hunting_selectivity + log_population);
371
372 return exp(
373 hunting_selectivity
374 + log(hunting_quota)
375 + log(2.0)
376 - 2.0 * log(t_birth_to_end_hunt)
377 - epsilon_h * hunting_effort_sd
378 - log_denominator
