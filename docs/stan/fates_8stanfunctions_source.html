53row_vector allocate_fates(
54 row_vector expected_fate_counts,
55 row_vector fate_allocation_noise,
56 real demographic_population_count,
57 int hunting_active_sweden,
58 int hunting_active_finland
59) {
60 // Active category dimensions and positions in paper order
61 int n_active_fates = 2 + hunting_active_sweden + hunting_active_finland;
62 array[n_active_fates] int active_fate_indices;
63 int next_active_fate = 2;
64
65 // Expected counts and adjusted concentrations
66 vector[n_active_fates] expected_active_fate_counts;
67 vector[n_active_fates] concentration_adjusted;
68 real min_expected_count;
69
70 // Log-ratio means and structured covariance factor
71 vector[n_active_fates - 1] mean_logratio;
72 vector[n_active_fates - 1] diagonal_variance;
73 real shared_variance;
74 matrix[n_active_fates - 1, n_active_fates - 1] fate_allocation_cholesky =
75 rep_matrix(0.0, n_active_fates - 1, n_active_fates - 1);
76
77 // Active allocations and full four-category output
78 vector[n_active_fates] logits;
79 vector[n_active_fates] active_fate_counts;
80 row_vector[4] realized_fate_counts = rep_row_vector(0.0, 4);
81
82 // Check dimensions and the population count before allocating
83 if (num_elements(expected_fate_counts) != 4) {
84 reject("allocate_fates requires four expected fate counts");
85 }
86 if (num_elements(fate_allocation_noise) < n_active_fates - 1) {
87 reject("allocate_fates requires at least ", n_active_fates - 1,
88 " innovations, received ", num_elements(fate_allocation_noise));
89 }
90 if (!(demographic_population_count > 0.0)) {
91 reject("allocate_fates requires a positive population count; received ",
92 demographic_population_count);
93 }
94
95 // Select survival, active hunting categories, and other death or bycatch
96 active_fate_indices[1] = 1;
97 if (hunting_active_sweden) {
98 active_fate_indices[next_active_fate] = 2;
99 next_active_fate += 1;
100 }
101 if (hunting_active_finland) {
102 active_fate_indices[next_active_fate] = 3;
103 next_active_fate += 1;
104 }
105 active_fate_indices[n_active_fates] = 4;
106 expected_active_fate_counts = expected_fate_counts[active_fate_indices]';
107 min_expected_count = min(expected_active_fate_counts);
108 if (!(min_expected_count > 0.0)) {
109 reject("allocate_fates requires strictly positive active expected counts; received ",
110 expected_active_fate_counts);
111 }
112
113 // Adjust concentrations without forming a potentially overflowing reciprocal
114 concentration_adjusted = expected_active_fate_counts
115 + expected_active_fate_counts / min_expected_count;
116
117 // Calculate log-ratio means and variances relative to survival
118 mean_logratio = digamma(concentration_adjusted[2:n_active_fates])
119 - rep_vector(digamma(concentration_adjusted[1]), n_active_fates - 1);
120 diagonal_variance = trigamma(concentration_adjusted[2:n_active_fates]);
121 shared_variance = trigamma(concentration_adjusted[1]);
122
123 // Factor diag(diagonal_variance) + shared_variance * 1 * 1' directly
124 // This avoids losing small diagonal terms when the shared variance is large
125 for (i in 1:(n_active_fates - 1)) {
126 real pivot_variance = diagonal_variance[i] + shared_variance;
127 fate_allocation_cholesky[i, i] = sqrt(pivot_variance);
128 if (i < n_active_fates - 1) {
129 for (j in (i + 1):(n_active_fates - 1)) {
130 fate_allocation_cholesky[j, i] = shared_variance / fate_allocation_cholesky[i, i];
131 }
132 // A bounded ratio avoids cancellation and overflow in the variance update
133 shared_variance *= diagonal_variance[i] / pivot_variance;
134 }
135 }
136
137 // Apply the same innovations and convert log-ratios into population counts
138 logits[1] = 0.0;
139 logits[2:n_active_fates] = mean_logratio
140 + fate_allocation_cholesky * fate_allocation_noise[1:(n_active_fates - 1)]';
141 active_fate_counts = demographic_population_count * softmax(logits);
142
143 // Return counts in paper order with inactive hunting categories left at zero
144 for (i in 1:n_active_fates) {
145 realized_fate_counts[active_fate_indices[i]] = active_fate_counts[i];
146 }
147
148 return realized_fate_counts;
149}
