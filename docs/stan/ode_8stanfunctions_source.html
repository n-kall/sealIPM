51vector hunting_count_derivative(
52 real tau,
53 vector H,
54 real n0,
55 real k,
56 real E_1,
57 real E_2,
58 real mu
59) {
60 real alive =
61 n0
62 * exp(
63 -(E_1 + E_2)
64 * (k * tau - square(tau) / 2.0)
65 - mu * tau
66 );
67
68 real dH =
69 alive
70 * E_1
71 * (k - tau);
72
73 return rep_vector(dH, 1);
74}

171vector compute_fate_probabilities(
172 data real t_birth_to_end_hunt,
173 real hunting_pressure_sweden,
174 real hunting_pressure_finland,
175 real mu_m,
176 data real rel_tol,
177 data real abs_tol,
178 data int max_num_steps,
179 data real probability_tolerance
180) {
181 vector[4] probability;
182 real probability_sum;
183 real k = t_birth_to_end_hunt;
184 vector[1] initial_hunting = rep_vector(0.0, 1);
185 array[1] real times = { k };
186
187 if (!(k >= 0.0 && k <= 1.0)) {
188 reject(
189 "compute_fate_probabilities requires t_birth_to_end_hunt in [0, 1]; ",
190 "received ", k
191 );
192 }
193 if (
194 !(hunting_pressure_sweden >= 0.0)
195 || !(hunting_pressure_finland >= 0.0)
196 || !(mu_m >= 0.0)
197 ) {
198 reject(
199 "compute_fate_probabilities requires non-negative hazards; ",
200 "hunting_pressure_sweden=", hunting_pressure_sweden,
201 ", hunting_pressure_finland=", hunting_pressure_finland,
202 ", mu_m=", mu_m
203 );
204 }
205
206 // Original Swedish hunting solve; exact zero for an inactive country.
207 probability[2] = 0.0;
208 if (k > 0.0 && hunting_pressure_sweden > 0.0) {
209 array[1] vector[1] hunted_sweden = ode_rk45_tol(
210 hunting_count_derivative, initial_hunting, 0.0, times,
211 rel_tol, abs_tol, max_num_steps,
212 1.0, k, hunting_pressure_sweden, hunting_pressure_finland, mu_m
213 );
214 probability[2] = hunted_sweden[1][1];
215 }
216
217 // Original Finnish hunting solve, with focal and competing pressures swapped.
218 probability[3] = 0.0;
219 if (k > 0.0 && hunting_pressure_finland > 0.0) {
220 array[1] vector[1] hunted_finland = ode_rk45_tol(
221 hunting_count_derivative, initial_hunting, 0.0, times,
222 rel_tol, abs_tol, max_num_steps,
223 1.0, k, hunting_pressure_finland, hunting_pressure_sweden, mu_m
224 );
225 probability[3] = hunted_finland[1][1];
226 }
227
228 // Calculate survival and remaining non-hunting death probability
229 probability[1] =
230 exp(-(hunting_pressure_sweden + hunting_pressure_finland) * square(k) / 2.0)
231 * exp(-mu_m);
232 probability[4] =
233 1.0 - probability[1] - probability[2] - probability[3];
234
235 // Validate the probability simplex before correcting small numerical errors
236 probability_sum =
237 sum(probability);
238
239 if (!(probability_tolerance > 0.0)) {
240 reject(
241 "compute_fate_probabilities requires a positive tolerance; ",
242 "probability_tolerance=", probability_tolerance
243 );
244 }
245
246 /*
247 * The !(x >= limit) form also rejects NaN because comparisons involving
248 * NaN evaluate to false
249 */
250 if (!(min(probability) >= -probability_tolerance)) {
251 reject(
252 "Joint fate calculation returned a materially negative probability; ",
253 "probability=", probability,
254 ", tolerance=", probability_tolerance
255 );
256 }
257
258 if (
259 !(probability_sum >= 1.0 - probability_tolerance)
260 || !(probability_sum <= 1.0 + probability_tolerance)
261 ) {
262 reject(
263 "Joint fate calculation failed probability conservation; probability=",
264 probability,
265 ", sum=", probability_sum,
266 ", tolerance=", probability_tolerance
267 );
268 }
269
270 /*
271 * Clamp only small boundary violations that passed the tolerance check
272 */
273 for (fate_id in 1:num_elements(probability)) {
274 probability[fate_id] =
275 fmax(0.0, probability[fate_id]);
276 }
277
278 probability_sum =
279 sum(probability);
280
281 if (!(probability_sum > 0.0)) {
282 reject(
283 "Joint fate probabilities sum to zero or are non-finite; probability=",
284 probability
285 );
286 }
287
288 /*
289 * Restore an exact simplex after any tolerated boundary correction
290 */
291 return probability / probability_sum;
292}
